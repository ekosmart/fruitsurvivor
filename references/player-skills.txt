            lightningSkill() {
                // Chain lightning: find a primary target in cursor direction, then chain to up to 10 enemies.
                // 1) Select first enemy in a forward cone toward mouse
                const CONE_ANGLE = 0.9; // radians wide cone
                const CONE_DIST = 420; // max distance to look for first target
                const dir = this.angle;

                // helper to check if candidate is in cone
                function inCone(px, py, ex, ey, dir, coneAngle, maxDist) {
                    const dx = ex - px, dy = ey - py;
                    const d = Math.hypot(dx, dy);
                    if (d > maxDist) return false;
                    const a = Math.atan2(dy, dx);
                    const diff = Math.abs(Math.atan2(Math.sin(a - dir), Math.cos(a - dir)));
                    return diff <= coneAngle / 2;
                }

                // find candidates in cone and sort by distance (include bosses)
                const candidates = enemies
                    .filter(e => inCone(this.x, this.y, e.x, e.y, dir, CONE_ANGLE, CONE_DIST))
                    .map(e => ({ e, d: Math.hypot(e.x - this.x, e.y - this.y) }))
                    .sort((a, b) => a.d - b.d)
                    .map(o => o.e);

                if (candidates.length === 0) {
                    // fallback: try any nearby enemy in a small radius in front
                    const fallback = enemies
                        .map(e => ({ e, d: Math.hypot(e.x - this.x, e.y - this.y) }))
                        .filter(o => o.d < 220)
                        .sort((a,b)=>a.d-b.d)
                        .map(o=>o.e);
                    if (fallback.length === 0) return; // nothing to hit
                    candidates.push(...fallback);
                }

                // build chain list starting from first candidate
                const maxChain = 10;
                const chain = [];
                const used = new Set();
                if (candidates.length > 0) {
                    chain.push(candidates[0]);
                    used.add(candidates[0]);
                }

                // chain to nearest unused enemy iteratively
                while (chain.length < maxChain) {
                    const last = chain[chain.length - 1];
                    if (!last) break;
                    let nearest = null; let nd = Infinity;
                    enemies.forEach(en => {
                        if (used.has(en)) return;
                        const d = Math.hypot(en.x - last.x, en.y - last.y);
                        if (d < nd && d < 320) { nd = d; nearest = en; }
                    });
                    if (!nearest) break;
                    chain.push(nearest); used.add(nearest);
                }

                if (chain.length === 0) return;

                // Create visual effects for chain lightning between points
                for (let i = 0; i < chain.length; i++) {
                    const srcX = (i === 0) ? this.x : chain[i-1].x;
                    const srcY = (i === 0) ? this.y : chain[i-1].y;
                    const tx = chain[i].x;
                    const ty = chain[i].y;
                    // push a lightning visual effect between src and target
                    const seg = { x: srcX, y: srcY, targetX: tx, targetY: ty, life: 18, type: 'lightningbolt', segments: [] };
                    const segCount = 10;
                    const angle = Math.atan2(ty - srcY, tx - srcX);
                    for (let s=0;s<=segCount;s++){
                        const p = s/segCount;
                        const bx = srcX + (tx - srcX) * p;
                        const by = srcY + (ty - srcY) * p;
                        const zig = (Math.random()-0.5) * 26 * (1 - Math.abs(p-0.5)*2);
                        const perp = angle + Math.PI/2;
                        seg.segments.push({ x: bx + Math.cos(perp)*zig, y: by + Math.sin(perp)*zig });
                    }
                    effects.push(seg);
                }

                // Apply damage and small stun on each in chain, show hit effect
                chain.forEach((enemy, idx) => {
                    try { enemy.takeDamage(computeScaledDamage(FRUITS.LIGHTNING.damage, this)); } catch(e) {}
                    enemy.stunned = Math.max(enemy.stunned || 0, 30);
                    effects.push({ x: enemy.x, y: enemy.y, radius: 20, life: 18, type: 'lightninghit' });
                });
            }
            
            distanceToLine(px, py, x1, y1, x2, y2) {
                const A = px - x1;
                const B = py - y1;
                const C = x2 - x1;
                const D = y2 - y1;
                
                const dot = A * C + B * D;
                const lenSq = C * C + D * D;
                let param = -1;
                
                if (lenSq !== 0) param = dot / lenSq;
                
                let xx, yy;
                if (param < 0) {
                    xx = x1;
                    yy = y1;
                } else if (param > 1) {
                    xx = x2;
                    yy = y2;
                } else {
                    xx = x1 + param * C;
                    yy = y1 + param * D;
                }
                
                const dx = px - xx;
                const dy = py - yy;
                return Math.sqrt(dx * dx + dy * dy);
            }

            jackpotSkill() {
                // 777 Fever: spin a flashy slot machine above player's head
                const owner = this;
                // show slot machine telegraph (will animate using effect.draw)
                const slotEffect = { type: 'slotMachine', owner: owner, life: 60, spin: true, reels: [null, null, null] };
                effects.push(slotEffect);

                // Instead of fully random reels, guarantee a triple outcome chosen from a weighted set
                // This makes the skill more rewarding: you will always get one of the triple outcomes.
                // weights: 7 (rare), cherries (common), skull (uncommon)
                const tripleWeights = [ {sym: '7', w: 5}, {sym: '🍒', w: 8}, {sym: '💀', w: 3} ];

                function weightedPick(arr) {
                    let total = 0; for (let it of arr) total += it.w || 1;
                    let r = Math.random() * total;
                    for (let it of arr) { r -= it.w || 1; if (r <= 0) return it.sym; }
                    return arr[arr.length-1].sym;
                }

                // resolve spin after short delay to allow telegraph animation
                setTimeout(() => {
                    // pick which triple the spin will produce (guaranteed)
                    const forced = weightedPick(tripleWeights);
                    const reels = [forced, forced, forced];

                    // attach result to an effect so draw can display it briefly
                    effects.push({ type: 'slotResult', owner: owner, life: 90, reels: reels });

                    // Determine if triple match (should always be true now)
                    const triple = true;

                    if (reels[0] === '7') {
                        // JACKPOT: massive coin explosion + heavy AoE damage
                        const cx = owner.x; const cy = owner.y;
                        const radius = 320;
                        effects.push({ type: 'jackpotCoinBurst', x: cx, y: cy, radius: radius, life: 48, color: '#FFD700' });

                        // Damage enemies in radius heavily and spawn coins
                        enemies.forEach(e => {
                            if (!e || e.dead) return;
                            const d = Math.hypot(e.x - cx, e.y - cy);
                            if (d < radius) {
                                try { e.takeDamage(computeScaledDamage(FRUITS.JACKPOT.damage * 5, owner)); } catch (e) {}
                            }
                        });

                        // spawn many coin drops around player
                        for (let i = 0; i < 28; i++) {
                            const a = Math.random() * Math.PI * 2;
                            const r = 24 + Math.random() * 260;
                            const dx = cx + Math.cos(a) * r;
                            const dy = cy + Math.sin(a) * r;

                            const coin = new DroppedItem(dx, dy, 'coin', coinVal);
                            // give a short random life offset so they scatter
                            coin.life = 600 + Math.floor(Math.random() * 300);
                            droppedItems.push(coin);
                        }

                        setShopMessage && setShopMessage('JACKPOT! 777 — Massive Payout!');
                    } else if (reels[0] === '🍒') {
                        // Cherry triple: potion rain, full heal, AND fruit rain (random fruit drops)
                        const cx = owner.x; const cy = owner.y;
                        effects.push({ type: 'potionRain', x: cx, y: cy - 180, life: 120, color: '#FF69B4' });
                        effects.push({ type: 'fruitRain', x: cx, y: cy - 220, life: 140, color: '#FFB7DA' });

                        // spawn potion drops falling from above
                        for (let i = 0; i < 12; i++) {
                            const px = cx + (Math.random() - 0.5) * 360;
                            const py = cy - 260 + Math.random() * 120;
                            const healAmt = Math.min(60 + Math.floor(Math.random() * 60), owner.maxHp);
                            const pot = new DroppedItem(px, py, 'potion', healAmt);
                            droppedItems.push(pot);
                        }

                        // spawn random fruit drops (fruit rain)
                        const fruitKeys = Object.keys(FRUITS).filter(k => FRUITS[k] && FRUITS[k].category !== 'bossfruit');
                        // prefer mid/rare fruits by simple filter (optional): keep set as-is for variety

                        for (let i = 0; i < fruitCount; i++) {
                            const fk = fruitKeys[Math.floor(Math.random() * fruitKeys.length)];

                            // slightly longer life so player has time to reach them

                    } else if (triple && reels[0] === '💀') {

            prismSkill() {
                // Prism (reworked): fire 7 colorful beams; each beam chains up to 7 enemies.
                const beamCount = 7;
                const beamChainMax = 7; // per-beam chain length
                const colors = ['#ff3b30','#ff9500','#ffd60a','#34c759','#007aff','#5856d6','#af52de']; // ROYGBIV-ish

                // sort enemies by distance from player (used to pick initial targets)
                const sorted = enemies
                    .map(e => ({ e, d: Math.hypot(e.x - this.x, e.y - this.y) }))
                    .sort((a, b) => a.d - b.d)
                    .map(o => o.e);

                if (sorted.length === 0) {
                    // No enemies: fire decorative beams forward and impact on ground
                    for (let i = 0; i < beamCount; i++) {
                        const angle = this.angle + (i / beamCount - 0.5) * 0.12 + (Math.random() - 0.5) * 0.03;
                        const tx = this.x + Math.cos(angle) * 420;
                        const ty = this.y + Math.sin(angle) * 420;
                        const color = colors[i % colors.length];
                        effects.push({ type: 'prismBeamLine', sx: this.x, sy: this.y, tx: tx, ty: ty, life: 18, color: color });
                        effects.push({ type: 'prismimpact', x: tx, y: ty, life: 20, color: color, radius: 24 });
                    }
                    setShopMessage && setShopMessage('Prismatic Barrage!');
                    return;
                }

                // Fire each beam. Each beam will chain to up to `beamChainMax` enemies.
                for (let i = 0; i < beamCount; i++) {
                    const initial = sorted[i % sorted.length];
                    if (!initial) continue;
                    const color = colors[i % colors.length];

                    // Build chain for this beam (per-beam used set so beams can overlap targets)
                    const chain = [];
                    const used = new Set();
                    chain.push(initial);
                    used.add(initial);

                    while (chain.length < beamChainMax) {
                        const last = chain[chain.length - 1];
                        if (!last) break;
                        let nearest = null; let nd = Infinity;
                        enemies.forEach(en => {
                            if (used.has(en)) return;
                            const d = Math.hypot(en.x - last.x, en.y - last.y);
                            // limit linkage distance so chain looks natural
                            if (d < nd && d < 360) { nd = d; nearest = en; }
                        });
                        if (!nearest) break;
                        chain.push(nearest); used.add(nearest);
                    }

                    // Apply visuals and damage along the chain (source -> first target -> next ...)
                    for (let j = 0; j < chain.length; j++) {
                        const srcX = (j === 0) ? this.x : chain[j - 1].x;
                        const srcY = (j === 0) ? this.y : chain[j - 1].y;
                        const tx = chain[j].x;
                        const ty = chain[j].y;

                        // beam segment visual
                        effects.push({ type: 'prismBeamLine', sx: srcX, sy: srcY, tx: tx, ty: ty, life: 20, color: color });
                        effects.push({ type: 'prismimpact', x: tx, y: ty, life: 22, color: color, radius: 28 });

                        // damage + small stagger per hit
                        try { chain[j].takeDamage(computeScaledDamage(FRUITS.PRISM.damage, this)); } catch (e) {}
                        chain[j].stunned = Math.max(chain[j].stunned || 0, 12);
                    }
                }

                setShopMessage && setShopMessage('Prismatic Barrage! — beams chain up to 7 enemies each');
            }

                // control up to 10 enemies (prioritize bosses/elites then distance)

                    effects.push({ x: target.x, y: target.y, radius: target.radius + 8, life: 60 * 7, type: 'puppetAura', color: '#000000' });

                }
